<script>
  const diccionarioLSA = {
    "beneficio": "Beneficio.mp4",
    "cebra": "Animales/Cebra.mp4",
    "lechuza": "Animales/Lechuza o Buho.mp4",
    "buho": "Animales/Lechuza o Buho.mp4",
    "vibora": "Animales/Vibora y Serpiente.mp4",
    "serpiente": "Animales/Vibora y Serpiente.mp4",
    "adquirir": "Contextos/Adquirir.mp4",
    "sindrome": "Salud/Sindrome.mp4",
    "cambiar": "Verbos/Verbos Cambiar.mp4",
    "atacar": "Vocabularios/Atacar.mp4",
    "cereales": "Vocabularios/Cereales.mp4",
    "conquistar": "Vocabularios/Conquistar.mp4",
    "creencia": "Vocabularios/Creencia.mp4",
    "dinosaurio": "Vocabularios/Dinosaurio.mp4",
    "dragon": "Vocabularios/Dragon.mp4",
    "durazno": "Vocabularios/Durazno.mp4",
    "encuesta": "Vocabularios/Encuesta.mp4",
    "frutilla": "Vocabularios/Frutilla.mp4",
    "idioma portugues": "Vocabularios/Idioma portugues.mp4",
    "itaembe mini": "Vocabularios/Itaembe Mini.mp4",
    "mascota": "Vocabularios/Mascota.mp4",
    "molino": "Vocabularios/Molino.mp4",
    "moraleja": "Vocabularios/Moraleja.mp4",
    "oferta": "Vocabularios/Oferta.mp4",
    "ofrecer": "Vocabularios/Ofrecer.mp4",
    "origen": "Vocabularios/Origen.mp4",
    "original": "Vocabularios/Original.mp4",
    "oruga": "Vocabularios/Oruga.mp4",
    "paleontologo": "Vocabularios/Panleontologo.mp4",
    "portugal": "Vocabularios/Portugal.mp4",
    "presentimiento": "Vocabularios/Presentimiento.mp4",
    "subsidio": "Vocabularios/Subcidio.mp4",
    "talento": "Vocabularios/Talento.mp4",
    "tero": "Vocabularios/Tero.mp4",
    "tolerancia": "Vocabularios/Tolerancia.mp4",
    "triunfo": "Vocabularios/Triunfo.mp4",
    "urquiza": "Vocabularios/Urquiza.mp4",
    "vencio": "Vocabularios/Vencio.mp4",
    "wasli": "Vocabularios/WASLI.mp4"
  };

  const reproductor = document.getElementById("reproductor");
  const estado = document.getElementById("estado");
  const inputTexto = document.getElementById("texto");
  const listaVocabulario = document.getElementById("listaVocabulario");

  reproductor.loop = true;

  for (let palabra in diccionarioLSA) {
    let boton = document.createElement("span");
    boton.className = "chip";
    boton.textContent = palabra;
    boton.onclick = () => {
      inputTexto.value = palabra;
      buscarSena();
    };
    listaVocabulario.appendChild(boton);
  }

  function normalizar(texto) {
    return texto.toLowerCase().trim().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
  }

  function buscarSena() {
    const consulta = normalizar(inputTexto.value);

    if (diccionarioLSA[consulta]) {
      reproductor.src = encodeURI(diccionarioLSA[consulta]);
      reproductor.play().catch(() => {});
      estado.style.color = "#0277bd";
      estado.textContent = "Mostrando seña: " + consulta;
    } else {
      reproductor.removeAttribute("src");
      reproductor.load();
      estado.style.color = "#c62828";
      estado.textContent = "No se encontró esa palabra en el diccionario.";
    }
  }

  reproductor.onerror = () => {
    if (reproductor.getAttribute("src")) {
      estado.style.color = "#c62828";
      estado.textContent = "Error: No se pudo cargar el archivo de video.";
    }
  };

  inputTexto.addEventListener("input", () => {
    const filtro = normalizar(inputTexto.value);
    const chips = listaVocabulario.querySelectorAll(".chip");
    chips.forEach(chip => {
      const coincide = normalizar(chip.textContent).includes(filtro);
      chip.style.display = coincide ? "inline-block" : "none";
    });
  });

  inputTexto.addEventListener("keypress", (e) => {
    if (e.key === "Enter") buscarSena();
  });
</script>
